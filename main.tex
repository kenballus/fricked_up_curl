\begin{frame}[fragile]{fuzzing}
    \begin{itemize}
        \pause \item How might we automate the process of finding bugs?
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{a simple fuzzing algorithm}
    \begin{enumerate}
        \pause \item generate an input
        \pause \item give it to the program under test (PUT)
        \pause \item if it crashes, save the input as a result
        \pause \item goto 1
    \end{enumerate}

    \begin{itemize}
        \pause \item This is known as a ``black box" fuzzing algorithm, because the only information collected from the PUT is whether it crashed.
        \pause \item What are some flaws with this approach?
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{a better fuzzing algorithm}
    \begin{enumerate}
        \pause \item begin with a few starting inputs (a ``corpus")
        \pause \item pick one of them
        \pause \item mutate it (add/remove/replace/increment/decrement bytes, flip bits, etc.)
        \pause \item give it to the program under test (PUT)
        \pause \item if it crashes, save the input as a result
        \pause \item otherwise, collect ``feedback" from the PUT
        \pause \item if the feedback is ``interesting," add this mutated input to the corpus
        \pause \item goto 2
    \end{enumerate}

    \begin{itemize}
        \pause \item This is known as a ``grey box" fuzzing algorithm, because we're collecting feedback from the PUT to inform input selection.
        \pause \item This is known as a ``generational" or ``evolutionary" fuzzing algorithm because inputs are mutated, then selected for survival (addition to the corpus) based on an evaluation of fitness (whether the feedback is interesting).
    \end{itemize}
\end{frame}
